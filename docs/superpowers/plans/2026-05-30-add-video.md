**Goal:** Add a "+ 添加影片" button to the `/videos` page that opens a multi-step modal supporting auto mode (filename→code→scrape→translate→rename pipeline with SSE real-time progress) and manual mode (WEST/CN/DUMMY codes, user-filled metadata).

**Architecture:** Two-phase approach — file upload via multipart POST creates a job and saves files to a per-job temp directory within the selected volume; SSE stream endpoint reads from a per-job queue populated by a background thread running the 12-step pipeline. Manual mode is fully synchronous (upload → insert → form → finalize-rename).

# ── Job management ──────────────────────────────────────────────

@dataclass
class _Job:
    job_id: str
    queue: queue.Queue = field(default_factory=queue.Queue)
    status: str = 'running'
    created_at: float = field(default_factory=time.time)


_jobs: dict = {}
_jobs_lock = threading.Lock()


def _cleanup_old_jobs():
    while True:
        time.sleep(600)
        cutoff = time.time() - 3600
        with _jobs_lock:
            stale = [jid for jid, j in _jobs.items() if j.created_at < cutoff]
            for jid in stale:
                del _jobs[jid]


threading.Thread(target=_cleanup_old_jobs, daemon=True).start()
```

def api_max_code(code_type: str):
    ctype = code_type.upper()
    if ctype not in ('WEST', 'CN', 'DUMMY'):
        return jsonify({'error': 'invalid code_type'}), 400
    conn = sqlite3.connect(DB_PATH)
    try:
        cursor = conn.cursor()
        cursor.execute(
            "SELECT code FROM videos WHERE UPPER(code) LIKE ?",
            (f'{ctype}-%',),
        )
        rows = cursor.fetchall()
    finally:
        conn.close()
    max_num = 0
    for (code,) in rows:
        try:
            num = int(code.split('-', 1)[1])
            if num > max_num:
                max_num = num
        except (ValueError, IndexError):
            continue
    return jsonify({'code_type': ctype, 'max_number': max_num, 'suggested': max_num + 1})

@app.post('/api/videos/add-manual')
def api_add_manual():
    volume_path_str = (request.form.get('volume_path') or '').strip()
    volume_path = Path(volume_path_str)
    if volume_path not in [Path(p) for p in MEDIA_LIBRARIES]:
        return jsonify({'error': 'invalid volume_path'}), 400

    code_type = (request.form.get('code_type') or '').strip().upper()
    if code_type not in ('WEST', 'CN', 'DUMMY'):
        return jsonify({'error': 'invalid code_type'}), 400

    try:
        code_number = int(request.form.get('code_number') or 0)
    except ValueError:
        return jsonify({'error': 'invalid code_number'}), 400
    if code_number < 1:
        return jsonify({'error': 'code_number must be >= 1'}), 400

    is_multipart = request.form.get('is_multipart') == '1'
    files = request.files.getlist('files[]')
    if not files:
        return jsonify({'error': 'no files'}), 400

    code = f'{code_type}-{code_number:03d}'
    job_id = str(uuid.uuid4())
    tmp_dir = volume_path / '.upload_tmp' / job_id
    tmp_dir.mkdir(parents=True, exist_ok=True)

    sorted_files = sorted(files, key=lambda f: secure_filename(f.filename) or '')
    scanned = []
    for i, f in enumerate(sorted_files):
        safe_name = secure_filename(f.filename) or f'file_{i}.bin'
        tmp_path = tmp_dir / safe_name
        f.save(str(tmp_path))

        part = f'part{i + 1}' if is_multipart else None
        record = VideoRecord(
            id=0, code=code, part=part,
            original_file_path=tmp_path,
            title_zh_short=None, title_zh=None, title_jp=None, actors=[],
        )
        dest_path = build_destination(record, dest_root=volume_path)
        dest_path.parent.mkdir(parents=True, exist_ok=True)
        tmp_path.rename(dest_path)

        st = dest_path.stat()
        scanned.append({
            'code': code,
            'part': part,
            'original_path': dest_path,
            'file_size': st.st_size,
            'file_mtime': datetime.fromtimestamp(st.st_mtime).isoformat(),
            'file_birthtime': datetime.fromtimestamp(
                getattr(st, 'st_birthtime', st.st_mtime)
            ).isoformat(),
        })

    try:
        tmp_dir.rmdir()
    except OSError:
        pass

    batch_insert_scanned_videos(scanned)
    sync_derived_video_tags_for_code(code)

    return jsonify({'codes': [code]})

@app.post('/api/videos/<string:code>/finalize-rename')
def api_finalize_rename(code: str):
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    try:
        cursor = conn.cursor()
        cursor.execute(
            "SELECT id, code, part, original_file_path, title_zh_short, title_zh, title_jp "
            "FROM videos WHERE UPPER(code) = UPPER(?) AND deleted = 0",
            (code,),
        )
        rows = cursor.fetchall()
        if not rows:
            return jsonify({'error': 'not found'}), 404

        results = []
        for row in rows:
            cursor.execute(
                "SELECT COALESCE(a.name_zh, a.name) AS name "
                "FROM actors a JOIN video_actor_link val ON val.actor_id = a.id "
                "WHERE val.video_id = ? ORDER BY a.id ASC",
                (row['id'],),
            )
            actors = [r['name'] for r in cursor.fetchall() if r['name']]
            record = VideoRecord(
                id=row['id'], code=row['code'], part=row['part'],
                original_file_path=Path(row['original_file_path']),
                title_zh_short=row['title_zh_short'],
                title_zh=row['title_zh'],
                title_jp=row['title_jp'],
                actors=actors,
            )
            new_name = build_filename(record)
            old_path = Path(row['original_file_path'])
            new_path = old_path.parent / new_name
            if old_path != new_path and old_path.exists():
                old_path.rename(new_path)
                cursor.execute(
                    "UPDATE videos SET original_file_path = ?, updated_at = CURRENT_TIMESTAMP "
                    "WHERE id = ?",
                    (str(new_path), row['id']),
                )
            results.append({'old': str(old_path), 'new': str(new_path)})

        conn.commit()
    finally:
        conn.close()

    return jsonify({'ok': True, 'new_paths': [r['new'] for r in results]})
```

def api_add_auto():
    volume_path_str = (request.form.get('volume_path') or '').strip()
    volume_path = Path(volume_path_str)
    if volume_path not in [Path(p) for p in MEDIA_LIBRARIES]:
        return jsonify({'error': 'invalid volume_path'}), 400

    files = request.files.getlist('files[]')
    if not files:
        return jsonify({'error': 'no files'}), 400

    job_id = str(uuid.uuid4())
    tmp_dir = volume_path / '.upload_tmp' / job_id
    tmp_dir.mkdir(parents=True, exist_ok=True)

    saved_paths = []
    for f in files:
        safe_name = secure_filename(f.filename) or f'file_{len(saved_paths)}.bin'
        dest = tmp_dir / safe_name
        f.save(str(dest))
        saved_paths.append(dest)

    with _jobs_lock:
        job = _Job(job_id=job_id)
        _jobs[job_id] = job

    t = threading.Thread(
        target=_run_auto_pipeline,
        args=(job, saved_paths, volume_path),
        daemon=True,
    )
    t.start()
    return jsonify({'job_id': job_id})


@app.get('/api/jobs/<string:job_id>/stream')
def api_job_stream(job_id: str):
    with _jobs_lock:
        job = _jobs.get(job_id)
    if not job:
        return jsonify({'error': 'not found'}), 404

    def generate():
        while True:
            try:
                event = job.queue.get(timeout=30)
            except queue.Empty:
                yield 'data: {"step":"ping","status":"running","message":""}\n\n'
                continue
            yield f'data: {json.dumps(event, ensure_ascii=False)}\n\n'
            if event.get('step') in ('done', 'error'):
                break

    return Response(
        generate(),
        mimetype='text/event-stream',
        headers={'Cache-Control': 'no-cache', 'X-Accel-Buffering': 'no'},
    )

def _scrape_with_playwright(base_code: str) -> dict:
    """Run Playwright scrape in the calling thread. Returns result_data dict or None."""
    from playwright.sync_api import sync_playwright
    from playwright_stealth import Stealth
    from core.scraper import scrape_video_info, load_javdb_cookies
    with sync_playwright() as p:
        browser = p.chromium.launch(
            headless=True, args=['--disable-blink-features=AutomationControlled']
        )
        context = browser.new_context(viewport={'width': 1280, 'height': 800})
        load_javdb_cookies(context)
        page = context.new_page()
        Stealth().apply_stealth_sync(page)
        try:
            return scrape_video_info(page, base_code)
        finally:
            browser.close()

        const router = VueRouter.useRouter();

        // ── Shared state ────────────────────────────────────────
        const addStep    = ref('file-select');
        const addFiles   = ref([]);
        const addVolume  = ref('');
        const addMode    = ref('auto');
        const addDoneCode = ref('');
        const addError   = ref('');

        const mediaLibraries = ref([]);
        const addFileInput = ref(null);

        // ── Auto mode state ─────────────────────────────────────
        const AUTO_STEPS = [
          'parse_code','calc_dir','upload','db_insert','tags',
          'scrape','actors','translate','short_title',
          'calc_filename','rename','db_update',
        ];
        const autoProgressSteps = ref([]);
        const autoFinished = ref(false);

        const autoProgressPct = computed(() => {
          const done = autoProgressSteps.value.filter(s => s.status === 'done').length;
          return Math.round((done / AUTO_STEPS.length) * 100);
        });

        function autoStepLabel(step) {
          const map = {
            parse_code: '解析番号', calc_dir: '计算目标目录', upload: '上传文件',
            db_insert: '写入数据库', tags: '同步标签', scrape: '刮削影片信息',
            actors: '获取演员信息', translate: '翻译标题', short_title: '计算短标题',
            calc_filename: '计算文件名', rename: '重命名文件', db_update: '更新数据库',
          };
          return map[step] || step;
        }

        // ── Manual mode state ────────────────────────────────────
        const manualCodeType    = ref('WEST');
        const manualCodeNumber  = ref(1);
        const manualMaxNumber   = ref(0);
        const manualIsMultipart = ref(false);
        const manualUploading   = ref(false);
        const manualCode        = ref('');
        const manualTitleJp     = ref('');
        const manualTitleZh     = ref('');
        const manualTitleShort  = ref('');
        const manualActors      = ref('');
        const manualTags        = ref([]);
        const manualSaving      = ref(false);
        const manualMetaError   = ref('');

        const addFileSorted = computed(() =>
          [...addFiles.value].sort((a, b) => a.name.localeCompare(b.name))
        );

        // ── Lifecycle ────────────────────────────────────────────
        onMounted(async () => {
          const libs = await apiFetch('/api/config/media-libraries');
          if (libs) mediaLibraries.value = libs;
        });

        // ── Handlers ─────────────────────────────────────────────
        function onAddFilesChange(e) {
          addFiles.value = Array.from(e.target.files || []);
          if (addFiles.value.length > 1) manualIsMultipart.value = true;
        }

        async function fetchMaxCode() {
          const data = await apiFetch(`/api/videos/max-code/${manualCodeType.value}`);
          if (data) {
            manualMaxNumber.value = data.max_number;
            manualCodeNumber.value = data.suggested;
          }
        }

        function startAddFlow() {
          if (addMode.value === 'auto') {
            startAutoUpload();
          } else {
            fetchMaxCode();
            addStep.value = 'manual-code';
          }
        }

        async function startAutoUpload() {
          addStep.value = 'auto-progress';
          autoFinished.value = false;
          addError.value = '';
          autoProgressSteps.value = AUTO_STEPS.map(s => ({ step: s, status: 'pending', message: '' }));

          const fd = new FormData();
          addFileSorted.value.forEach(f => fd.append('files[]', f));
          fd.append('volume_path', addVolume.value);

          let jobId;
          try {
            const res = await fetch('/api/videos/add-auto', { method: 'POST', body: fd });
            const data = await res.json();
            if (!res.ok) throw new Error(data.error || '上传失败');
            jobId = data.job_id;
          } catch (e) {
            addError.value = e.message;
            autoFinished.value = true;
            return;
          }

          const es = new EventSource(`/api/jobs/${jobId}/stream`);
          es.onmessage = (e) => {
            const event = JSON.parse(e.data);
            if (event.step === 'ping') return;
            const idx = autoProgressSteps.value.findIndex(s => s.step === event.step);
            if (idx >= 0) {
              autoProgressSteps.value[idx] = { ...autoProgressSteps.value[idx], ...event };
            } else {
              autoProgressSteps.value.push(event);
            }
            if (event.step === 'done') {
              addDoneCode.value = event.code || '';
              autoFinished.value = true;
              addStep.value = 'done';
              es.close();
            } else if (event.step === 'error') {
              addError.value = event.message;
              autoFinished.value = true;
              es.close();
            }
          };
          es.onerror = () => {
            addError.value = '连接中断';
            autoFinished.value = true;
            es.close();
          };
        }

        async function startManualUpload() {

          const fd = new FormData();
          addFileSorted.value.forEach(f => fd.append('files[]', f));
          fd.append('volume_path', addVolume.value);

          fd.append('code_number', String(manualCodeNumber.value));
          fd.append('is_multipart', manualIsMultipart.value ? '1' : '0');
          try {

            manualCode.value = data.codes[0];

          const code = manualCode.value;
          const body = {

          try {

            addDoneCode.value = code;
            addStep.value = 'done';
          } catch (e) {
            manualMetaError.value = e.message;
          } finally {
            manualSaving.value = false;
          }
        }
